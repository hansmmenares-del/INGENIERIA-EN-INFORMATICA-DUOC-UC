EMPRESA
    - #* idEmpresa [INTEGER]
    - * nombre [VARCHAR]


MAQUINARIA
    - #* idUnico [VARCHAR]
    - * marca [VARCHAR]
    - * modelo [VARCHAR]
    - o anioDeFabricacion [INTEGER]
    - * estadoActual [VARCHAR] 
        (DISPONIBLE, EN_MANTENCION, ARRENDADA)

    MOTORIZADA
        - #* idUnico [PK, FK -> MAQUINARIA]
        - * potenciaHp [NUMERIC]

    MANUAL
        - #* idUnico [PK, FK -> MAQUINARIA]
        - * pesoKg [NUMERIC]


EMPLEADO
    - #* run [VARCHAR]
    - * nombres [VARCHAR]
    - * apellidoPaterno [VARCHAR]
    - * apellidoMaterno [VARCHAR]
    - * cargo [VARCHAR]
    - * areaDeTrabajo [VARCHAR]

    OPERARIO
        - #* run [PK, FK -> EMPLEADO]
        - * tipoDeLicencia [VARCHAR]
            (A2, A3, A4, A5)
        - * numeroDeLicencia [VARCHAR]
        - * fechaEmisionDocumentoLicencia [DATE]
        - * fechaVencimientoDocumentoLicencia [DATE]

    SUPERVISORDETERRENO
        - #* run [PK, FK -> EMPLEADO]


CLIENTE
    - #* idCliente [INTEGER]

    INDEPENDIENTE
        - #* idCliente [PK, FK -> CLIENTE]
        - #* run [VARCHAR] [UK]
        - * primerNombre [VARCHAR]
        - o segundoNombre [VARCHAR]
        - * apellidoPaterno [VARCHAR]
        - * apellidoMaterno [VARCHAR]
        - * direccion [VARCHAR]
        - * comuna [VARCHAR]
        - * region [VARCHAR]

    COOPERATIVAAGRICOLA
        - #* idCliente [PK, FK -> CLIENTE]
        - #* rut [VARCHAR] [UK]
        - * razonSocial [VARCHAR]
        - * direccion [VARCHAR]
        - * comuna [VARCHAR]
        - * region [VARCHAR]


TELEFONO
    - #* idTelefono [INTEGER]
    - * numero [VARCHAR]
    - * tipo [VARCHAR]


AUTORIZACION
    - #* idAutorizacion [INTEGER]
    - * fechaInicio [DATE]
    - * fechaTermino [DATE]
    - * runOperario [FK -> OPERARIO]
    - * idCooperativa [FK -> COOPERATIVAAGRICOLA]


ARRIENDO
    - #* idArriendo [INTEGER]
    - * fechaHoraInicioDelArriendo [DATE]
    - * fechaHoraTerminoDelArriendo [DATE]
    - * estadoTecnicoInicialHorometro [NUMERIC]
    - * estadoTecnicoInicialCombustible [NUMERIC]
    - o estadoTecnicoFinalHorometro [NUMERIC]
    - o estadoTecnicoFinalCombustible [NUMERIC]
    - o costoArriendo [DECIMAL] [DERIVADO]
    - * tipoFacturacion [VARCHAR]
        (CLIENTE, COOPERATIVA)
    - * idCliente [FK -> CLIENTE]
    - * idMaquinaria [FK -> MAQUINARIA]
    - * runSupervisor [FK -> SUPERVISORDETERRENO]
    - o runOperario [FK -> OPERARIO]
    - o idCooperativaFacturacion [FK -> COOPERATIVAAGRICOLA]


ESTADOARRIENDO
    - #* idEstadoArriendo [INTEGER]
    - * tipoDeEstado [VARCHAR]
        (EN_CREACION, CONFIRMADO, EN_TERRENO, FINALIZADO, CANCELADO)
    - * fechaInicio [DATE]
    - o fechaTermino [DATE]
    - * idArriendo [FK -> ARRIENDO]


COMPAÑIAASEGURADORA
    - #* idCompaniaAseguradora [INTEGER]
    - * razonSocial [VARCHAR]
    - * rut [VARCHAR] [UK]


SEGUROAGRICOLA
    - #* idSeguro [INTEGER]
    - * nombre [VARCHAR]
    - o descripcion [VARCHAR]
    - * idCompaniaAseguradora [FK -> COMPAÑIAASEGURADORA]


ARRIENDOSEGURO
    - #* idArriendo [PK, FK -> ARRIENDO]
    - #* idSeguro [PK, FK -> SEGUROAGRICOLA]


PROVEEDOREXTERNOKIT
    - #* idProveedor [INTEGER]
    - * razonSocial [VARCHAR]
    - * rut [VARCHAR] [UK]
    - * direccion [VARCHAR]
    - * comuna [VARCHAR]
    - * region [VARCHAR]


ELEMENTOKIT
    - #* idElemento [INTEGER]
    - * nombre [VARCHAR]
    - * tipo [VARCHAR]
        (COMBUSTIBLE, LUBRICANTE, HERRAMIENTA)
    - o descripcion [VARCHAR]
    - * idProveedor [FK -> PROVEEDOREXTERNOKIT]


DETALLEKITARRIENDO
    - #* idArriendo [PK, FK -> ARRIENDO]
    - #* idElemento [PK, FK -> ELEMENTOKIT]
    - * cantidad [NUMERIC]
